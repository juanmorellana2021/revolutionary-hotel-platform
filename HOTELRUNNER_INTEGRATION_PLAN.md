// File: classes/HotelRunnerAPI.php

class HotelRunnerAPI {
    private $apiKey;
    private $apiSecret;
    private $baseUrl = 'https://api.hotelrunner.com/v1'; // Example URL
    
    public function __construct() {
        $this->apiKey = getenv('HOTELRUNNER_API_KEY');
        $this->apiSecret = getenv('HOTELRUNNER_API_SECRET');
    }
    
    private function makeRequest($endpoint, $method = 'GET', $data = null) {
        $ch = curl_init();
        $url = $this->baseUrl . $endpoint;
        
        $headers = [
            'Authorization: Bearer ' . $this->getAuthToken(),
            'Content-Type: application/json'
        ];
        
        curl_setopt($ch, CURLOPT_URL, $url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_HTTPHEADER, $headers);
        
        if ($method === 'POST' || $method === 'PUT') {
            curl_setopt($ch, CURLOPT_CUSTOMREQUEST, $method);
            curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));
        }
        
        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);
        
        return [
            'code' => $httpCode,
            'data' => json_decode($response, true)
        ];
    }
    
    // Create Property
    public function createProperty($hotelData) {
        return $this->makeRequest('/properties', 'POST', [
            'name' => $hotelData['hotel_name'],
            'address' => $hotelData['address'],
            'city' => $hotelData['city'],
            'country' => 'PE', // Peru
            'phone' => $hotelData['phone'],
            'email' => $hotelData['email'],
            'rooms' => $hotelData['total_rooms']
        ]);
    }
    
    // Update Availability
    public function updateAvailability($hrPropertyId, $roomTypeId, $date, $available) {
        return $this->makeRequest('/inventory/availability', 'PUT', [
            'property_id' => $hrPropertyId,
            'room_type_id' => $roomTypeId,
            'date' => $date,
            'available' => $available
        ]);
    }
    
    // Update Rates
    public function updateRate($hrPropertyId, $roomTypeId, $date, $rate) {
        return $this->makeRequest('/inventory/rates', 'PUT', [
            'property_id' => $hrPropertyId,
            'room_type_id' => $roomTypeId,
            'date' => $date,
            'rate' => $rate,
            'currency' => 'USD'
        ]);
    }
    
    // Get Bookings
    public function getBookings($hrPropertyId, $fromDate, $toDate) {
        return $this->makeRequest(
            "/bookings?property_id={$hrPropertyId}&from={$fromDate}&to={$toDate}"
        );
    }
}

**Sync Service:**

```php
// File: classes/HRSyncService.php

class HRSyncService {
    private $api;
    private $db;
    
    public function __construct() {
        $this->api = new HotelRunnerAPI();
        $this->db = Database::getInstance();
    }
    
    // Activate hotel on HotelRunner
    public function activateHotel($hotelId) {
        $hotel = $this->getHotel($hotelId);
        
        // Create property in HotelRunner
        $response = $this->api->createProperty($hotel);
        
        if ($response['code'] === 201) {
            $hrPropertyId = $response['data']['property_id'];
            
            // Save to database
            $this->db->query(
                "UPDATE hotels SET hr_property_id = ?, hr_active = TRUE 
                 WHERE id = ?",
                [$hrPropertyId, $hotelId]
            );
            
            // Sync rooms
            $this->syncRooms($hotelId);
            
            // Push initial availability (next 365 days)
            $this->syncAvailability($hotelId, 365);
            
            return [
                'success' => true,
                'property_id' => $hrPropertyId
            ];
        }
        
        return ['success' => false, 'error' => $response['data']];
    }
    
    // Sync room availability to HotelRunner
    public function syncAvailability($hotelId, $days = 30) {
        $hotel = $this->getHotel($hotelId);
        
        if (!$hotel['hr_active']) {
            return ['error' => 'Hotel not active on HotelRunner'];
        }
        
        $rooms = $this->getRooms($hotelId);
        $startDate = date('Y-m-d');
        
        $synced = 0;
        for ($i = 0; $i < $days; $i++) {
            $date = date('Y-m-d', strtotime("+$i days"));
            
            foreach ($rooms as $room) {
                $available = $this->getAvailableRoomCount(
                    $hotelId, 
                    $room['id'], 
                    $date
                );
                
                $this->api->updateAvailability(
                    $hotel['hr_property_id'],
                    $room['hr_room_type_id'],
                    $date,
                    $available
                );
                
                $synced++;
            }
        }
        
        // Update last sync time
        $this->db->query(
            "UPDATE hotels SET hr_last_sync = NOW() WHERE id = ?",
            [$hotelId]
        );
        
        return ['synced' => $synced];
    }
    
    // Import booking from HotelRunner
    public function importBooking($webhookData) {
        $hrReservationId = $webhookData['reservation_id'];
        $hrPropertyId = $webhookData['property_id'];
        
        // Find hotel by HotelRunner Property ID
        $hotel = $this->db->query(
            "SELECT * FROM hotels WHERE hr_property_id = ?",
            [$hrPropertyId]
        )->fetch();
        
        if (!$hotel) {
            $this->logError('Hotel not found', $webhookData);
            return false;
        }
        
        // Check if booking already exists
        $existing = $this->db->query(
            "SELECT id FROM hr_channel_bookings 
             WHERE hr_reservation_id = ?",
            [$hrReservationId]
        )->fetch();
        
        if ($existing) {
            return ['already_exists' => true];
        }
        
        // Create booking in AiniTravel PMS
        $bookingId = $this->createBooking([
            'hotel_id' => $hotel['id'],
            'guest_name' => $webhookData['guest_name'],
            'guest_email' => $webhookData['guest_email'],
            'guest_phone' => $webhookData['guest_phone'],
            'check_in' => $webhookData['check_in'],
            'check_out' => $webhookData['check_out'],
            'total_amount' => $webhookData['total_amount'],
            'source' => 'OTA',
            'booking_status' => 'confirmed'
        ]);
        
        // Track as channel booking
        $this->db->query(
            "INSERT INTO hr_channel_bookings 
             (hotel_id, booking_id, hr_reservation_id, channel_name, 
              channel_reservation_id, commission_amount, commission_percent)
             VALUES (?, ?, ?, ?, ?, ?, ?)",
            [
                $hotel['id'],
                $bookingId,
                $hrReservationId,
                $webhookData['channel'], // 'Booking.com', 'Expedia', etc.
                $webhookData['channel_reservation_id'],
                $webhookData['commission_amount'],
                $webhookData['commission_percent']
            ]
        );
        
        return ['success' => true, 'booking_id' => $bookingId];
    }
}
```

**Webhook Endpoint:**

```php
// File: api/hotelrunner_webhook.php

require_once '../db_connection.php';
require_once '../classes/HRSyncService.php';

// Verify webhook signature (HotelRunner will provide method)
$signature = $_SERVER['HTTP_X_HOTELRUNNER_SIGNATURE'] ?? '';
$payload = file_get_contents('php://input');

if (!verifyWebhookSignature($payload, $signature)) {
    http_response_code(403);
    exit('Invalid signature');
}

$data = json_decode($payload, true);

$syncService = new HRSyncService();

switch ($data['event_type']) {
    case 'booking.created':
        $result = $syncService->importBooking($data);
        break;
        
    case 'booking.modified':
        $result = $syncService->modifyBooking($data);
        break;
        
    case 'booking.cancelled':
        $result = $syncService->cancelBooking($data);
        break;
        
    default:
        $result = ['error' => 'Unknown event type'];
}

header('Content-Type: application/json');
echo json_encode($result);
```
